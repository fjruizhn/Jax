def load_at(name, revision, tempdir):
    source = subprocess.run(
        ["git", "show", f"{revision}:{SCANNER}"], capture_output=True, check=True
    ).stdout
    path = Path(tempdir) / f"{name}.py"
    path.write_bytes(source)
    spec = importlib.util.spec_from_file_location(name, path)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module


def detect(scanner, kind, source):

def documented_whitelist_case(kind, source):
    if kind == "shell":
        tokens = source.replace("'", " ").replace('"', " ").split()
        return any(
            tokens[i] == "docker" and tokens[i + 1] in WHITELIST and tokens[i + 2] == "rm"
            for i in range(len(tokens) - 2)
        )
    try:
        tree = ast.parse(source)
    except SyntaxError:
        return False
    return any(
        isinstance(node, (ast.List, ast.Tuple))
        and any(
            i >= 2
            and isinstance(node.elts[i - 2], ast.Constant)
            and node.elts[i - 2].value == "docker"
            and isinstance(node.elts[i - 1], ast.Constant)
            and node.elts[i - 1].value in WHITELIST
            and isinstance(node.elts[i], ast.Constant)
            and node.elts[i].value == "rm"
            for i in range(len(node.elts))
        )
        for node in ast.walk(tree)
    )


RONDA5_MEJORAS = {
    # El segundo `rm` de la lista (docker) ahora se evalúa aunque el primero no sea docker.
    '["ssh", "rm", "docker", "rm", "-f", n]',
    '["sudo", "-u", "rm", "docker", "rm", "-f", n]',
}


def required_improvement(kind, source):
    if kind == "python" and (source in RONDA5_MEJORAS or source in RONDA6_MEJORAS):
        return True
    if "container remove" in source or '"container", "remove"' in source:
        return True  # ronda 7: `docker container remove` es el alias de `docker container rm`
    if "--workdir" in source and "compose" in source:
        return True
    if kind == "shell":
        return "docker compose -f a -f b -f c -f d -f e rm" in source
    try:
        tree = ast.parse(source)
    except SyntaxError:
        return False
    sequence = ["docker", "compose", "-f", "a", "-f", "b", "-f", "c",
                "-f", "d", "-f", "e", "rm"]
    return any(
        isinstance(node, (ast.List, ast.Tuple))
        and any(
            [elt.value for elt in node.elts[i:i + len(sequence)]] == sequence
            for i in range(len(node.elts) - len(sequence) + 1)
            if all(isinstance(elt, ast.Constant) and isinstance(elt.value, str)
                   for elt in node.elts[i:i + len(sequence)])
        )
        for node in ast.walk(tree)
    )

with tempfile.TemporaryDirectory(prefix="jax357-differential-") as tempdir:
    master = load_at("master_scanner", ARGS.master_sha, tempdir)
    candidate = load_at("candidate_scanner", ARGS.candidate_sha, tempdir)
    mismatches = []
    exceptions = []
    expected_improvements = []
    master_marked = candidate_marked = 0
    cases = [(kind, source, "pr") for kind, source in positive + negative]
    cases += [(item["kind"], item["source"], item["origin"]) for item in auditor_cases]
    cases += [(kind, source, "codex-30") for kind, source in own_cases]
    for kind, source, origin in cases:
        master_result = detect(master, kind, source)
        candidate_result = detect(candidate, kind, source)
        required = required_improvement(kind, source)
        master_marked += master_result
        candidate_marked += candidate_result
        if master_result and not candidate_result and documented_whitelist_case(kind, source):
            status = "WHITELIST"
            exceptions.append((origin, kind, source))
        elif master_result and not candidate_result:
            status = "LEAK"
            mismatches.append((origin, kind, source, status))
        elif required and not candidate_result:
            status = "EXPECTATION_MISS"
            mismatches.append((origin, kind, source, status))
        elif candidate_result and not master_result and required:
            status = "EXPECTED_NEW_MARK"
            expected_improvements.append((origin, kind, source))
        elif candidate_result and not master_result:
            status = "NEW_MARK"
            mismatches.append((origin, kind, source, status))
        else:
            status = "MATCH"
        print(f"{status}\t{origin}\t{kind}\t{source!r}")
    print(f"master_sha={ARGS.master_sha} candidate_sha={ARGS.candidate_sha}")
    print(f"pr_cases={len(positive)+len(negative)} auditor_cases={len(auditor_cases)} own_cases={len(own_cases)} total={len(cases)}")
    print(f"master_marked={master_marked} candidate_marked={candidate_marked}")
    print(f"leaks={sum(item[3] == 'LEAK' for item in mismatches)} unexpected_new_marks={sum(item[3] == 'NEW_MARK' for item in mismatches)} expectation_misses={sum(item[3] == 'EXPECTATION_MISS' for item in mismatches)} expected_candidate_only_marks={len(expected_improvements)} whitelist_exceptions={len(exceptions)}")
    for origin, kind, source, status in mismatches:
        print(f"MISMATCH {status} {origin} {kind}: {source!r}")
    for origin, kind, source in exceptions:
        print(f"DOCUMENTED_WHITELIST {origin} {kind}: {source!r}")
    if mismatches:
        raise SystemExit(1)
