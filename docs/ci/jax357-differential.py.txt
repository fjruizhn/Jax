def load_at(name, revision, tempdir):
    source = subprocess.run(
        ["git", "show", f"{revision}:{SCANNER}"], capture_output=True, check=True
    ).stdout
    path = Path(tempdir) / f"{name}.py"
    path.write_bytes(source)
    spec = importlib.util.spec_from_file_location(name, path)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module


def detect(scanner, kind, source):
    return bool(scanner.culpables_en_texto(source, es_python=(kind == "python")))

with tempfile.TemporaryDirectory(prefix="jax357-differential-") as tempdir:
    master = load_at("master_scanner", ARGS.master_sha, tempdir)
    candidate = load_at("candidate_scanner", ARGS.candidate_sha, tempdir)
    false_negatives = []
    false_positives = []
    master_false_positives = []
    master_marked = candidate_marked = 0
    for kind, source in positive + negative:
        expected = (kind, source) in positive
        master_result = detect(master, kind, source)
        candidate_result = detect(candidate, kind, source)
        master_marked += master_result
        candidate_marked += candidate_result
        if master_result and not expected:
            master_false_positives.append((kind, source))
        if candidate_result != expected:
            (false_negatives if expected else false_positives).append((kind, source))
    print(f"master_sha={ARGS.master_sha} candidate_sha={ARGS.candidate_sha}")
    print(f"positives={len(positive)} negatives={len(negative)} total={len(positive)+len(negative)}")
    print(f"master_marked={master_marked} candidate_marked={candidate_marked}")
    print(f"candidate_false_negatives={len(false_negatives)} candidate_false_positives={len(false_positives)}")
    print(f"master_false_positives={len(master_false_positives)}")
    for kind, source in master_false_positives:
        print(f"DOCUMENTED_MASTER_FALSE_POSITIVE {kind}: {source}")
    for kind, source in false_negatives + false_positives:
        print(f"CANDIDATE_MISMATCH {kind}: {source}")
    if false_negatives or false_positives:
        raise SystemExit(1)
